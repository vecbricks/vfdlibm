import java.util.concurrent.atomic.AtomicLong;
import java.util.stream.LongStream;

public class Proto {
  static final double LN2_HI = 6.93147180369123816490e-01;
  static final double LN2_LO = 1.90821492927058770002e-10;
  static final double TWO54 = 1.80143985094819840000e+16;
  static final double LP1 = 6.666666666666735130e-01;
  static final double LP2 = 3.999999999940941908e-01;
  static final double LP3 = 2.857142874366239149e-01;
  static final double LP4 = 2.222219843214978396e-01;
  static final double LP5 = 1.818357216161805012e-01;
  static final double LP6 = 1.531383769920937332e-01;
  static final double LP7 = 1.479819860511658591e-01;
  static final double ZERO = 0.0;

  static int hi(double d) {
    return (int) (Double.doubleToRawLongBits(d) >>> 32);
  }

  static double withHi(double y, int hi) {
    return Double.longBitsToDouble(
        ((long) hi << 32) | (Double.doubleToRawLongBits(y) & 0xffffffffL));
  }

  static double cLog1p(double x, boolean generalPathOnly) {
    double hfsq;
    double f = 0;
    double c = 0;
    double s;
    double z;
    double r;
    double u;
    int k;
    int hx;
    int hu = 0;
    int ax;
    hx = hi(x);
    ax = hx & 0x7fffffff;
    k = 1;
    if (hx < 0x3FDA827A) {
      if (ax >= 0x3ff00000) {
        if (x == -1.0) {
          return -TWO54 / ZERO;
        } else {
          return (x - x) / (x - x);
        }
      }
      if (ax < 0x3e200000) {
        if (TWO54 + x > ZERO && ax < 0x3c900000) {
          return x;
        } else {
          return x - x * x * 0.5;
        }
      }
      if (hx > 0 || hx <= 0xbfd2bec3) {
        k = 0;
        f = x;
        hu = 1;
      }
    }
    if (hx >= 0x7ff00000) {
      return x + x;
    }
    if (k != 0) {
      if (hx < 0x43400000) {
        u = 1.0 + x;
        hu = hi(u);
        k = (hu >> 20) - 1023;
        c = (k > 0) ? 1.0 - (u - x) : x - (u - 1.0);
        c /= u;
      } else {
        u = x;
        hu = hi(u);
        k = (hu >> 20) - 1023;
        c = 0;
      }
      hu &= 0x000fffff;
      if (hu < 0x6a09e) {
        u = withHi(u, hu | 0x3ff00000);
      } else {
        k += 1;
        u = withHi(u, hu | 0x3fe00000);
        hu = (0x00100000 - hu) >> 2;
      }
      f = u - 1.0;
    }
    hfsq = 0.5 * f * f;
    if (hu == 0) {
      if (f == ZERO) {
        if (k == 0) {
          return ZERO;
        } else {
          c += k * LN2_LO;
          return k * LN2_HI + c;
        }
      }
      if (!generalPathOnly) {
        r = hfsq * (1.0 - 0.66666666666666666 * f);
        if (k == 0) {
          return f - r;
        } else {
          return k * LN2_HI - ((r - (k * LN2_LO + c)) - f);
        }
      }
    }
    s = f / (2.0 + f);
    z = s * s;
    r = z * (LP1 + z * (LP2 + z * (LP3 + z * (LP4 + z * (LP5 + z * (LP6 + z * LP7))))));
    if (k == 0) {
      return f - (hfsq - s * (hfsq + r));
    } else {
      return k * LN2_HI - ((hfsq - (s * (hfsq + r) + (k * LN2_LO + c))) - f);
    }
  }

  public static void main(String[] args) {
    long from = Long.parseUnsignedLong(args[0], 16);
    long count = Long.parseLong(args[1]);
    int threads = Integer.parseInt(args[2]);
    System.setProperty("java.util.concurrent.ForkJoinPool.common.parallelism", args[2]);
    long chunk = 1L << 22;
    AtomicLong diffs = new AtomicLong();
    AtomicLong shortPath = new AtomicLong();
    AtomicLong notOracle = new AtomicLong();
    AtomicLong printed = new AtomicLong();
    long t0 = System.nanoTime();
    LongStream.range(0, (count + chunk - 1) / chunk).parallel().forEach(ci -> {
      long lo = from + ci * chunk;
      long hiB = Math.min(from + count, lo + chunk);
      long d = 0;
      long sp = 0;
      long no = 0;
      for (long b = lo; b < hiB; b++) {
        double x = Double.longBitsToDouble(b);
        long a = Double.doubleToRawLongBits(cLog1p(x, false));
        long g = Double.doubleToRawLongBits(cLog1p(x, true));
        if (a != g) {
          d++;
          if (printed.incrementAndGet() <= 16) {
            System.out.printf("differs at %016x: C %016x, general %016x, StrictMath %016x%n", b, a,
                g, Double.doubleToRawLongBits(StrictMath.log1p(x)));
          }
        }
        double u = 1.0 + x;
        int hu = hi(u) & 0xfffff;
        if ((hu == 0 || hu > 0xffffc) && u != withHi(0.0, hi(u))) {
          sp++;
        }
        if ((b & 1023) == 0 && a != Double.doubleToRawLongBits(StrictMath.log1p(x))) {
          no++;
        }
      }
      diffs.addAndGet(d);
      shortPath.addAndGet(sp);
      notOracle.addAndGet(no);
    });
    System.out.printf("from %016x, %d inputs, %d on the short path, %d differ, %d of the sampled"
        + " not StrictMath's, %.1f s on %d threads%n", from, count, shortPath.get(), diffs.get(),
        notOracle.get(), (System.nanoTime() - t0) / 1e9, threads);
  }
}
